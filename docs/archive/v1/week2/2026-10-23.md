## 구현 (80분)

1. 저장소 인터페이스 (`src/lib/subscriptions/repository.ts`)
```ts
import type { Subscription } from '@/domain/types';
import type { SubscriptionInput } from './schema';

export interface SubscriptionRepository {
  list(): Promise<Subscription[]>;
  findById(id: string): Promise<Subscription | undefined>;
  create(input: SubscriptionInput): Promise<Subscription>;
  update(id: string, input: SubscriptionInput): Promise<Subscription | undefined>;
  remove(id: string): Promise<boolean>;
}
```

2. 인메모리 구현 (`src/lib/subscriptions/memory-repository.ts`)
```ts
import type { Subscription } from '@/domain/types';
import type { SubscriptionInput } from './schema';
import type { SubscriptionRepository } from './repository';

export function createMemoryRepository(): SubscriptionRepository {
  const store = new Map<string, Subscription>();

  return {
    async list() {
      return [...store.values()];
    },
    async findById(id) {
      return store.get(id);
    },
    async create(input) {
      const sub: Subscription = { id: crypto.randomUUID(), ...input };
      store.set(sub.id, sub);
      return sub;
    },
    async update(id, input) {
      if (!store.has(id)) return undefined;
      const sub: Subscription = { id, ...input };
      store.set(id, sub);
      return sub;
    },
    async remove(id) {
      return store.delete(id);
    },
  };
}
```

3. 앱에서 쓸 싱글턴 (`src/lib/subscriptions/index.ts`)
```ts
import { createMemoryRepository } from './memory-repository';

export const subscriptionRepository = createMemoryRepository();
```

4. Route Handler (`src/app/api/subscriptions/route.ts`)
```ts
import { NextResponse } from 'next/server';
import { subscriptionInputSchema } from '@/lib/subscriptions/schema';
import { subscriptionRepository } from '@/lib/subscriptions';

export async function GET() {
  const items = await subscriptionRepository.list();
  return NextResponse.json({ items });
}

export async function POST(request: Request) {
  const body = await request.json().catch(() => null);
  const parsed = subscriptionInputSchema.safeParse(body);

  if (!parsed.success) {
    const fields = Object.fromEntries(
      parsed.error.issues.map((i) => [i.path[0] ?? 'form', i.message]),
    );
    return NextResponse.json(
      { error: { code: 'VALIDATION_ERROR', message: '입력값을 확인해 주세요', fields } },
      { status: 400 },
    );
  }

  const created = await subscriptionRepository.create(parsed.data);
  return NextResponse.json(created, { status: 201 });
}
```

## 학습 포인트
- Next.js 개발 서버는 핫 리로드 때 모듈 상태가 초기화될 수 있다. 인메모리 저장소는 개발 중 데이터가 사라질 수 있음을 알아두고, 필요하면 `globalThis`에 붙여서 유지한다 (4주차 E2E 준비에 활용)
- `request.json()`은 잘못된 JSON에서 예외를 던지므로 `catch(() => null)`로 400 처리 경로에 합친다
