## 구현 (90분)

1. 컴포넌트 (`src/components/subscriptions/SubscriptionForm.tsx`)
```tsx
'use client';

import { useState, type FormEvent } from 'react';
import { subscriptionInputSchema, type SubscriptionInput } from '@/lib/subscriptions/schema';

type Errors = Partial<Record<keyof SubscriptionInput, string>>;

export function SubscriptionForm({ onSubmit }: { onSubmit: (v: SubscriptionInput) => Promise<void> }) {
  const [values, setValues] = useState({ name: '', amount: '', cycle: 'monthly', startDate: '' });
  const [errors, setErrors] = useState<Errors>({});
  const [submitting, setSubmitting] = useState(false);

  async function handleSubmit(e: FormEvent<HTMLFormElement>) {
    e.preventDefault();
    const parsed = subscriptionInputSchema.safeParse({
      ...values,
      amount: Number(values.amount),
    });
    if (!parsed.success) {
      const next: Errors = {};
      for (const issue of parsed.error.issues) {
        next[issue.path[0] as keyof SubscriptionInput] = issue.message;
      }
      setErrors(next);
      return;
    }
    setErrors({});
    setSubmitting(true);
    try {
      await onSubmit(parsed.data);
    } finally {
      setSubmitting(false);
    }
  }

  return (
    <form onSubmit={handleSubmit} noValidate>
      <div>
        <label htmlFor="name">서비스명</label>
        <input
          id="name"
          name="name"
          value={values.name}
          onChange={(e) => setValues({ ...values, name: e.target.value })}
          aria-invalid={Boolean(errors.name)}
          aria-describedby={errors.name ? 'name-error' : undefined}
        />
        {errors.name && <p id="name-error" role="alert">{errors.name}</p>}
      </div>

      {/* amount, cycle, startDate 필드도 같은 패턴으로 작성 */}

      <button type="submit" disabled={submitting}>
        {submitting ? '저장 중…' : '저장'}
      </button>
    </form>
  );
}
```

## 기본기 노트 (30분): React 렌더링 트리거와 상태 업데이트

**state 업데이트는 즉시 바뀌지 않는다.** 다음 렌더링에서 반영된다.
```tsx
const [count, setCount] = useState(0);

function handleClick() {
  setCount(count + 1);
  setCount(count + 1);
  console.log(count); // 여전히 이전 값
}
// 클릭 한 번에 count는 1만 증가한다 (두 호출이 같은 스냅샷 값을 사용)
```
해결: 함수형 업데이트
```tsx
setCount((c) => c + 1);
setCount((c) => c + 1); // 2 증가
```

정리할 것:
- 렌더링 트리거: state 변경, 부모 렌더링, Context 값 변경
- 렌더링과 DOM 반영은 다르다. 컴포넌트 함수가 호출되는 것(렌더)과 DOM이 바뀌는 것(커밋)을 구분한다
- 메모이제이션(`memo`, `useMemo`, `useCallback`)은 비용이 큰 계산이나 자식 props 안정화가 필요할 때만 쓴다
