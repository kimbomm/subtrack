## 구현 (100분)

1. 테스트 먼저 작성 (`src/domain/filter.test.ts`)
```ts
import { describe, it, expect } from 'vitest';
import { filterSubscriptions } from './filter';
import type { Subscription } from './types';

const today = '2026-11-06';

const items: Subscription[] = [
  { id: 'a', name: '넷플릭스', amount: 17000, cycle: 'monthly', startDate: '2026-01-10' },
  { id: 'b', name: '연간 도메인', amount: 120000, cycle: 'yearly', startDate: '2026-11-01' },
  { id: 'c', name: '주간 세차', amount: 5000, cycle: 'weekly', startDate: '2026-11-02' },
];

describe('filterSubscriptions', () => {
  it('필터가 없으면 전체를 반환한다', () => {
    expect(filterSubscriptions(items, {}, today)).toHaveLength(3);
  });

  it('주기로 필터링한다', () => {
    const result = filterSubscriptions(items, { cycle: 'yearly' }, today);
    expect(result.map((s) => s.id)).toEqual(['b']);
  });

  it('7일 이내 갱신 항목만 남긴다', () => {
    // a: 다음 결제일 2026-12-10 (제외), b: 2026-11-01 시작 연간 → 다음 2027-11-01 (제외)
    // c: 2026-11-02 시작 주간 → 다음 2026-11-09 (포함, 3일 후)
    const result = filterSubscriptions(items, { dueWithinDays: 7 }, today);
    expect(result.map((s) => s.id)).toEqual(['c']);
  });

  it('원본 배열을 변경하지 않는다', () => {
    const copy = structuredClone(items);
    filterSubscriptions(items, { cycle: 'monthly' }, today);
    expect(items).toEqual(copy);
  });

  it('조건이 모두 맞지 않으면 빈 배열', () => {
    expect(filterSubscriptions(items, { cycle: 'weekly', dueWithinDays: 1 }, today)).toEqual([]);
  });
});
```
> 테스트의 기대값은 먼저 손으로 계산해서 적는다. 계산한 값이 틀리면 테스트가 아니라 계산을 다시 검토한다.

2. 구현 (`src/domain/filter.ts`)
```ts
import type { Subscription, BillingCycle } from './types';
import { nextBillingDate } from './billingDate';

export interface SubscriptionFilter {
  cycle?: BillingCycle;
  dueWithinDays?: number;
}

function daysBetween(from: string, to: string): number {
  const ms = Date.parse(`${to}T00:00:00Z`) - Date.parse(`${from}T00:00:00Z`);
  return Math.round(ms / 86_400_000);
}

export function filterSubscriptions(
  items: readonly Subscription[],
  filter: SubscriptionFilter,
  today: string,
): Subscription[] {
  return items.filter((s) => {
    if (filter.cycle && s.cycle !== filter.cycle) return false;
    if (filter.dueWithinDays !== undefined) {
      const next = nextBillingDate(s.startDate, s.cycle, today);
      if (daysBetween(today, next) > filter.dueWithinDays) return false;
    }
    return true;
  });
}
```

## 학습 포인트
- 순수 함수는 같은 입력에 같은 출력을 주고 외부 상태를 바꾸지 않는다. 테스트가 쉽고 재사용이 쉽다
- 현재 시각을 함수 안에서 직접 읽으면 테스트가 날짜에 따라 깨진다. 인자로 주입하는 것이 기본 원칙이다
